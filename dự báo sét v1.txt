end

local RainbowGradients = {}
local function ApplyRainbowBorder(parentObj, thickness)
thickness = thickness or 2
local stroke = Instance.new("UIStroke")
stroke.Thickness = thickness
stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
stroke.Color = Color3.fromRGB(255, 255, 255)
stroke.Parent = parentObj

local gradient = Instance.new("UIGradient")  
gradient.Color = ColorSequence.new({  
    ColorSequenceKeypoint.new(0.00, Color3.fromRGB(255, 0, 0)),  
    ColorSequenceKeypoint.new(0.16, Color3.fromRGB(255, 127, 0)),  
    ColorSequenceKeypoint.new(0.33, Color3.fromRGB(255, 255, 0)),  
    ColorSequenceKeypoint.new(0.50, Color3.fromRGB(0, 255, 0)),  
    ColorSequenceKeypoint.new(0.66, Color3.fromRGB(0, 150, 255)),  
    ColorSequenceKeypoint.new(0.83, Color3.fromRGB(139, 0, 255)),  
    ColorSequenceKeypoint.new(1.00, Color3.fromRGB(255, 0, 0))  
})  
gradient.Rotation = 0  
gradient.Parent = stroke  

table.insert(RainbowGradients, gradient)  
return stroke

end

RunService.RenderStepped:Connect(function()
local angle = (tick() * 180) % 360
for i = #RainbowGradients, 1, -1 do
local grad = RainbowGradients[i]
if grad and grad.Parent then
grad.Rotation = angle
else
table.remove(RainbowGradients, i)
end
end
end)

local function HopLowServer()
ShowDarkNotification(L("NotifHopSearch"))
local placeId = game.PlaceId
local serversUrl = "https://games.roblox.com/v1/games/" .. placeId .. "/servers/Public?sortOrder=Asc&limit=100"

local success, result = pcall(function()  
    return HttpService:JSONDecode(game:HttpGet(serversUrl))  
end)  

if success and result and result.data then  
    local targetServer  
    for _, server in ipairs(result.data) do  
        if server.playing and server.playing == 1 and server.id ~= game.JobId then  
            targetServer = server.id  
            break  
        end  
    end  

    if not targetServer then  
        for _, server in ipairs(result.data) do  
            if server.playing and server.playing < server.maxPlayers and server.id ~= game.JobId then  
                targetServer = server.id  
                break  
            end  
        end  
    end  

    if targetServer then  
        ShowDarkNotification(L("NotifHopping"))  
        TeleportService:TeleportToPlaceInstance(placeId, targetServer, LocalPlayer)  
    else  
        ShowDarkNotification(L("NotifNoServer"))  
    end  
else  
    ShowDarkNotification(L("NotifFailServer"))  
end

end

CreateButton(TabLocalPlayer, "HopBtn", function()
HopLowServer()
end)

local function IsNearPlayer(object, maxDistance)
maxDistance = maxDistance or Settings.MaxDistance
local Character = LocalPlayer.Character

if not Character or not Character:FindFirstChild("HumanoidRootPart") then  
    return true  
end  

local PlayerPos = Character.HumanoidRootPart.Position  
local ObjPos  

if object:IsA("BasePart") then  
    ObjPos = object.Position  
elseif object:IsA("Model") then  
    ObjPos = object.PrimaryPart and object.PrimaryPart.Position or object:GetPivot().Position  
elseif object:IsA("Attachment") then  
    ObjPos = object.WorldPosition  
end  

if ObjPos then  
    return (PlayerPos - ObjPos).Magnitude <= maxDistance  
end  

return true

end

local function IsEarlyLightningSignal(object)
local name = string.lower(object.Name)
local keywords = {
"lightning", "strike", "warning", "thunder", "spark", "cloud",
"indicator", "redcircle", "danger", "target", "charge", "aura",
"flash", "electric", "pre_strike", "spawn"
}

for _, word in ipairs(keywords) do  
    if string.find(name, word) then return true end  
end  

if object:IsA("ParticleEmitter") or object:IsA("Beam") or object:IsA("Highlight") or object:IsA("Sound") then  
    local parentName = string.lower(object.Parent and object.Parent.Name or "")  
    for _, word in ipairs(keywords) do  
        if string.find(parentName, word) then return true end  
    end  
end  

return false

end

local function FireHarvest(targetObj)
if not Settings.Enabled or not Settings.AutoHarvest then return end

local prompt  
if targetObj then  
    if targetObj:IsA("ProximityPrompt") then  
        prompt = targetObj  
    else  
        prompt = targetObj:FindFirstChildWhichIsA("ProximityPrompt", true)  
    end  
end  

if not prompt and LocalPlayer.Character then  
    local myPos = LocalPlayer.Character.HumanoidRootPart.Position  
    local minDist = Settings.MaxDistance  
    for _, v in ipairs(Workspace:GetDescendants()) do  
        if v:IsA("ProximityPrompt") and v.Enabled and v.Parent and v.Parent:IsA("BasePart") then  
            local dist = (myPos - v.Parent.Position).Magnitude  
            if dist <= minDist then  
                prompt = v  
                minDist = dist  
            end  
        end  
    end  
end  

if prompt and prompt.Enabled then  
    if Settings.BypassHold then prompt.HoldDuration = 0 end  
    task.spawn(function()  
        for _ = 1, Settings.AutoRetry do  
            fireproximityprompt(prompt)  
            task.wait(0.05)  
        end  
    end)  
    ShowDarkNotification(L("NotifSuccess"))  
end

end

local function TriggerActionSequence(sourceObj)
if State.IsProcessing then return end
State.IsProcessing = true
State.IsIdle = false

if Settings.HarvestDelay > 0 then  
    StatusLabel.Text = L("StatusWaiting", Settings.HarvestDelay)  
    task.wait(Settings.HarvestDelay)  
end  

if Settings.AutoHarvest then  
    FireHarvest(sourceObj)  
end  

StatusLabel.Text = L("StatusArrived")  
ShowDarkNotification(L("NotifProcessed"))  
task.wait(2)  

State.LightningHits = 0  
State.IsProcessing = false  
State.IsIdle = true  

if Settings.Enabled then  
    StatusLabel.Text = L("StatusIdle")  
end

end

local sweepConnection = nil
local loopConnection = nil

StartMonitoring = function()
if sweepConnection then sweepConnection:Disconnect() end
if loopConnection then task.cancel(loopConnection) end

sweepConnection = Workspace.DescendantAdded:Connect(function(child)  
    if not Settings.Enabled or State.IsProcessing then return end  

    if IsEarlyLightningSignal(child) and IsNearPlayer(child) then  
        State.LightningHits = State.LightningHits + 1  
          
        if State.LightningHits < 3 then  
            StatusLabel.Text = L("StatusHits", State.LightningHits)  
            return  
        end  
          
        if State.LightningHits >= 3 then  
            StatusLabel.Text = L("StatusHarvest", State.LightningHits)  
            task.spawn(function()  
                TriggerActionSequence(child)  
            end)  
        end  
    end  
end)  

loopConnection = task.spawn(function()  
    while Settings.Enabled do  
        task.wait(0.2)  
        if not State.IsProcessing then  
            for _, obj in ipairs(Workspace:GetChildren()) do  
                if IsEarlyLightningSignal(obj) and IsNearPlayer(obj) then  
                    State.LightningHits = State.LightningHits + 1  
                      
                    if State.LightningHits < 3 then  
                        StatusLabel.Text = L("StatusHits", State.LightningHits)  
                    elseif State.LightningHits >= 3 then  
                        StatusLabel.Text = L("StatusHarvest", State.LightningHits)  
                        task.spawn(function()  
                            TriggerActionSequence(obj)  
                        end)  
                    end  
                    task.wait(1)  
                    break  
                end  
            end  
        end  
    end  
end)

end

StopMonitoring = function()
if sweepConnection then sweepConnection:Disconnect() end
if loopConnection then task.cancel(loopConnection) end
StatusLabel.Text = L("StatusDisabled")
end

if Settings.Enabled then
StartMonitoring()
end
updateUI()
